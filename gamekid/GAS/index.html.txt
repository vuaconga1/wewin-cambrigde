    async function postApi_(action, payload) {
      try {
        const primaryResult = await requestApiWithRetries_(
          action,
          payload || {},
          API_URL_OVERRIDE
        );

        if (shouldUseLegacyFallback_(primaryResult)) {
          setLoadingMessage_('Hồ sơ đang chuyển đổi, đang đọc dữ liệu dự phòng...');
          return await jsonpApi_(action, payload || {}, 45000, API_URL_FALLBACK);
        }

        return primaryResult;
      } catch (primaryError) {
        if (!API_URL_FALLBACK || API_URL_FALLBACK === API_URL_OVERRIDE) {
          throw primaryError;
        }

        setLoadingMessage_('API mới đang bận, đang đọc dữ liệu dự phòng...');
        showNotice_(
          'Hệ thống đang tự chuyển sang nguồn dữ liệu dự phòng. Phụ huynh không cần bấm thêm.',
          'info'
        );
        try {
          return await jsonpApi_(action, payload || {}, 45000, API_URL_FALLBACK);
        } catch (fallbackError) {
          throw primaryError || fallbackError;
        }
      }
    }

    async function requestApiWithRetries_(action, payload, baseUrl) {
      let lastError = null;

      for (let attempt = 0; attempt < API_TIMEOUTS_MS.length; attempt++) {
        if (attempt > 0) {
          const retryNumber = attempt + 1;
          setLoadingMessage_(`Kết nối chưa ổn định, đang thử lại (${retryNumber}/${API_TIMEOUTS_MS.length})...`);
          showNotice_(
            `Kết nối tới hệ thống chưa ổn định. WeWIN đang tự thử lại (${retryNumber}/${API_TIMEOUTS_MS.length}), phụ huynh không cần bấm thêm.`,
            'info'
          );
          await delay_(API_RETRY_DELAYS_MS[attempt]);
        }

        try {
          const res = await jsonpApi_(action, payload || {}, API_TIMEOUTS_MS[attempt], baseUrl);

          if (res && res.ok === false && res.retryable) {
            const retryableError = new Error(res.message || 'Hệ thống đang bận.');
            retryableError.retryable = true;
            throw retryableError;
          }

          return res;
        } catch (err) {
          lastError = err;
          const isLastAttempt = attempt >= API_TIMEOUTS_MS.length - 1;
          if (isLastAttempt || !isTransientApiError_(err)) throw err;
        }
      }

      throw lastError || new Error('Không kết nối được tới hệ thống.');
    }

    function shouldUseLegacyFallback_(res) {
      const code = String(res && res.code ? res.code : '').toUpperCase();
      return Boolean(
        res &&
        res.ok === false &&
        (
          code === 'PROFILE_NOT_IMPORTED' ||
          code === 'PROFILE_DATA_NOT_READY' ||
          code === 'GAME_SSO_NOT_CONFIGURED'
        )
      );
    }

    function delay_(ms) {
      return new Promise(resolve => setTimeout(resolve, Math.max(0, Number(ms) || 0)));
    }

    function isTransientApiError_(err) {
      if (err && err.retryable) return true;

      const message = String(err && err.message ? err.message : err || '').toLowerCase();
      return [
        'phản hồi quá lâu',
        'không tải được api',
        'failed to load',
        'network',
        'timeout',
        'temporarily',
        'service unavailable',
        'server error'
      ].some(signal => message.includes(signal));
    }

    function createTransientApiError_(message) {
      const err = new Error(message);
      err.retryable = true;
      return err;
    }

    function jsonpApi_(action, payload, timeoutMs, baseUrlOverride) {
      return new Promise((resolve, reject) => {
        const baseUrl = (baseUrlOverride || API_URL_OVERRIDE || '').trim();

        if (!baseUrl) {
          reject(new Error('Thiếu API_URL_OVERRIDE.'));
          return;
        }

        const callbackName =
          '__eduxPortalCallback_' +
          Date.now() +
          '_' +
          Math.random().toString(36).slice(2);

        const params = new URLSearchParams();
        params.set('mode', 'api');
        params.set('action', action);
        params.set('callback', callbackName);

        Object.keys(payload || {}).forEach(key => {
          const value = payload[key];
          if (value !== undefined && value !== null) {
            params.set(key, String(value));
          }
        });

        const script = document.createElement('script');
        let completed = false;

        const cleanup = () => {
          if (completed) return;
          completed = true;

          try {
            delete window[callbackName];
          } catch (_) {
            window[callbackName] = undefined;
          }

          if (script && script.parentNode) {
            script.parentNode.removeChild(script);
          }
        };

        const timer = setTimeout(() => {
          cleanup();
          reject(createTransientApiError_('API phản hồi quá lâu. Hệ thống sẽ tự thử lại.'));
        }, Math.max(5000, Number(timeoutMs) || 30000));

        window[callbackName] = function (res) {
          clearTimeout(timer);
          cleanup();
          resolve(res);
        };

        script.onerror = function () {
          clearTimeout(timer);
          cleanup();
          reject(createTransientApiError_('Không tải được API. Hệ thống sẽ tự thử lại.'));
        };

        script.src = baseUrl + '?' + params.toString();
        document.head.appendChild(script);
      });
    }

    /* ── Game trắc nghiệm: tab mới + SSO JWT từ Apps Script ── */
    async function openQuizGameTab_() {
      const btn = document.getElementById('openQuizGameBtn');
      const sid = (
        (currentData && currentData.student && currentData.student.id) ||
        (currentAuth && currentAuth.studentId) ||
        ''
      ).toString().trim();
      const name = (
        (currentData && currentData.student && currentData.student.name) || ''
      ).toString().trim();
      const accessCode = (currentAuth && currentAuth.accessCode) || DEFAULT_ACCESS_CODE;

      if (!sid || sid === '--') {
        alert('Chưa có mã học viên. Vui lòng đăng nhập lại.');
        return;
      }

      const prevLabel = btn ? btn.querySelector('span') : null;
      const prevText = prevLabel ? prevLabel.textContent : '';
      if (btn) {
        btn.disabled = true;
        if (prevLabel) prevLabel.textContent = 'Đang mở...';
      }

      // Mở tab trắng ngay khi click (trước await). Không dùng noopener ở đây vì trình duyệt trả về null dù tab vẫn mở.
      const newTab = window.open('about:blank', '_blank');
      if (!newTab) {
        if (btn) {
          btn.disabled = false;
          if (prevLabel) prevLabel.textContent = prevText || 'Mở game trắc nghiệm';
        }
        alert('Trình duyệt chặn cửa sổ mới. Vui lòng cho phép popup cho trang này.');
        return;
      }
      try { newTab.opener = null; } catch (_) { }

      try {
        const res = await postApi_('game_sso_token', {
          studentId: sid,
          accessCode: accessCode,
          name: name && name !== '--' ? name : sid
        });
        if (!res || !res.ok || !res.url) {
          throw new Error((res && res.message) || 'Không tạo được phiên đăng nhập game');
        }
        newTab.location.href = res.url;
      } catch (err) {
        try { newTab.close(); } catch (_) { }
        alert(err && err.message ? err.message : 'Không mở được game trắc nghiệm');
      } finally {
        if (btn) {
          btn.disabled = false;
          if (prevLabel) prevLabel.textContent = prevText || 'Mở game trắc nghiệm';
        }
      }
    }


    /************************************************************
     * NORMALIZE API DATA
     *
     * API mới từ Apps Script đọc trực tiếp Drive ROOT và trả về:
     * {
     *   ok: true,
     *   student,
     *   profileFile,
     *   profile: { sheets: { OVERVIEW, DAILY, ATTENDANCE, SCORE, REPORT, LESSON... } },
     *   data: { student, profile }
     * }
     *
     * UI EDUX nguyên bản cần:
     * {
     *   student, feeInfo, metrics, dailyScores, attendance,
     *   baoBai, mocktests, reportText
     * }
     ************************************************************/
    function normalizePortalData_(res) {
      const raw = res || {};
      const data = raw.data || {};

      // Nếu API cũ đã trả đúng format frontend thì vẫn chuẩn hóa lại Daily data
      // để hỗ trợ điểm dạng 9/11, 11/11, 9.5 và nhận xét dạng text.
      if (data.metrics || data.dailyScores || data.attendance || data.baoBai || data.mocktests || data.reportText) {
        const normalizedData = Object.assign({}, data);
        normalizedData.dailyScores = normalizeDailyScoreItems_(data.dailyScores || []);

        const attendanceItems = Array.isArray(data.attendance) ? data.attendance : [];
        const metricFromDaily = buildMetrics_(attendanceItems, normalizedData.dailyScores);
        const recalculatedAttendanceMetrics = attendanceItems.length
          ? {
              attendancePresent: metricFromDaily.attendancePresent,
              attendanceAbsent: metricFromDaily.attendanceAbsent,
              attendanceExcused: metricFromDaily.attendanceExcused
            }
          : {};

        normalizedData.metrics = Object.assign(
          {},
          data.metrics || {},
          recalculatedAttendanceMetrics,
          {
            avgScore: metricFromDaily.avgScore,
            latestScore: metricFromDaily.latestScore
          }
        );

        return normalizedData;
      }

      const studentRaw = raw.student || data.student || {};
      const profile = raw.profile || data.profile || {};
      const sheets = profile.sheets || {};

      const overviewRows = getSheetRows_(sheets, 'OVERVIEW');
      const dailyRows = getSheetRows_(sheets, 'DAILY');
      const historyDailyRows = getSheetRows_(sheets, 'HISTORY_DAILY');
      const attendanceRows = getSheetRows_(sheets, 'ATTENDANCE');
      const historyAttendanceRows = getSheetRows_(sheets, 'HISTORY_ATTENDANCE');
      const lessonRows = getSheetRows_(sheets, 'LESSON');
      const historyRows = getSheetRows_(sheets, 'HISTORY');
      const scoreRows = getSheetRows_(sheets, 'SCORE');
      const reportRows = getSheetRows_(sheets, 'REPORT');

      const studentId =
        studentRaw.studentId ||
        studentRaw.id ||
        extractStudentCodeFromText_(raw.profileFile && raw.profileFile.name) ||
        currentAuth.studentId ||
        '';

      const student = {
        name:
          studentRaw.name ||
          studentRaw.studentName ||
          findLabelValue_(overviewRows, ['họ tên', 'tên học viên', 'student name', 'name']) ||
          studentId ||
          '--',

        id:
          studentId ||
          findLabelValue_(overviewRows, ['mã học viên', 'mã hv', 'student id', 'id']) ||
          '--',

        className:
          studentRaw.className ||
          findLabelValue_(overviewRows, ['lớp', 'class', 'class name']) ||
          '--',

        teacher:
          studentRaw.teacher ||
          findLabelValue_(overviewRows, ['giáo viên', 'teacher', 'tutor']) ||
          '--',

        dob:
          studentRaw.dob ||
          findLabelValue_(overviewRows, ['ngày sinh', 'sinh nhật', 'dob', 'birthday']) ||
          '--'
      };

      const feeInfo = {
        feeStatus:
          studentRaw.feeStatus ||
          findLabelValue_(overviewRows, ['tình hình báo phí', 'học phí', 'fee status']) ||
          '--',

        startDate:
          studentRaw.startDate ||
          findLabelValue_(overviewRows, ['ngày bắt đầu', 'start date']) ||
          '--',

        endDate:
          studentRaw.endDate ||
          findLabelValue_(overviewRows, ['ngày kết thúc', 'end date']) ||
          '--'
      };

      const dailyScores = mergeTimelineLiveFirst_(
        parseDailyScores_(dailyRows),
        parseDailyScores_(historyDailyRows)
      );
      const attendance = mergeTimelineLiveFirst_(
        parseAttendance_(attendanceRows),
        parseAttendance_(historyAttendanceRows)
      );
      const baoBai = buildBaoBaiTimeline_(lessonRows, historyRows);
      let mocktests = parseMocktests_(reportRows, student.id);
      if (!mocktests.length) mocktests = parseMocktests_(scoreRows, student.id);
      const reportText = parseReportText_(reportRows);
      const metrics = buildMetrics_(attendance, dailyScores);

      return {
        student,
        updatedAt: new Date().toLocaleString('vi-VN'),
        feeInfo,
        metrics,
        dailyScores,
        attendance,
        baoBai,
        mocktests,
        reportText,
        profileFile: raw.profileFile || null,
        rawProfile: profile
      };
    }

    function getSheetRows_(sheets, key) {
      const obj = sheets && sheets[key];
      if (!obj || !Array.isArray(obj.values)) return [];
      return obj.values || [];
    }

    function findLabelValue_(rows, labels) {
      const wanted = labels.map(normalizeVi_);

      for (let r = 0; r < rows.length; r++) {
        const row = rows[r] || [];

        for (let c = 0; c < row.length; c++) {
          const cell = cleanText_(row[c]);
          if (!cell) continue;

          const nCell = normalizeVi_(cell);
          const matched = wanted.some(label => nCell.includes(label));
          if (!matched) continue;

          for (let k = c + 1; k < row.length; k++) {
            const right = cleanText_(row[k]);
            if (right) return right;
          }

          for (let rr = r + 1; rr < Math.min(rows.length, r + 4); rr++) {
            const below = cleanText_((rows[rr] || [])[c]);
            if (below) return below;
          }
        }
      }

      return '';
    }

    function parseDailyScores_(rows) {
      const out = [];

      rows.forEach(row => {
        const cells = (row || []).map(cleanText_);
        const date = cells.find(isDateLike_);
        if (!date) return;

        const candidates = cells.filter(cell => cell !== date && isDailyScoreCell_(cell));
        if (!candidates.length) return;

        const withScore = candidates.filter(cell => parseDailyScoreText_(cell).scoreNumber !== '');
        const value = withScore.length ? withScore[withScore.length - 1] : candidates[candidates.length - 1];

        out.push(buildDailyScoreItem_(date, value));
      });

      return dedupeByDate_(out).slice(-30).reverse();
    }

    function buildDailyScoreItem_(date, value) {
      const text = cleanText_(value);
      const parsed = parseDailyScoreText_(text);

      return {
        date,
        rawScore: text,
        score: parsed.score || '--',
        scoreNumber: parsed.scoreNumber,
        scoreValue: parsed.scoreValue,
        scoreRaw: parsed.rawScore,
        scoreText: parsed.comment || text,
        comment: parsed.comment || text,
        displayText: text
      };
    }

    function normalizeDailyScoreItems_(items) {
      return (items || []).map(item => {
        const raw = cleanText_(item.rawScore || item.displayText || item.originalText || '');
        const sourceText = raw || cleanText_([item.score, item.scoreText || item.comment].filter(Boolean).join(' '));
        const parsed = parseDailyScoreText_(sourceText);

        const hasParsedScore = parsed.scoreNumber !== '';
        const scoreFromItem = cleanText_(item.score || '');
        const scoreNumberFromItem = cleanText_(item.scoreNumber || '');
        const normalizedScoreNumber = hasParsedScore
          ? parsed.scoreNumber
          : normalizeScoreNumberForAverage_(scoreNumberFromItem || scoreFromItem);

        return Object.assign({}, item, {
          rawScore: sourceText,
          displayText: sourceText,
          score: hasParsedScore ? parsed.score : (normalizedScoreNumber !== '' ? formatScoreForDisplay_(Number(normalizedScoreNumber)) : '--'),
          scoreNumber: hasParsedScore ? parsed.scoreNumber : normalizedScoreNumber,
          scoreValue: hasParsedScore ? parsed.scoreValue : (normalizedScoreNumber !== '' ? Number(normalizedScoreNumber) : ''),
          scoreRaw: hasParsedScore ? parsed.rawScore : scoreFromItem,
          scoreText: hasParsedScore ? parsed.comment : (cleanText_(item.scoreText || item.comment || sourceText)),
          comment: hasParsedScore ? parsed.comment : (cleanText_(item.comment || item.scoreText || sourceText))
        });
      });
    }

    function normalizeScoreNumberForAverage_(value) {
      const parsed = parseDailyScoreText_(value);
      if (parsed.scoreNumber !== '') return parsed.scoreNumber;

      const n = Number(cleanText_(value).replace(',', '.'));
      if (isNaN(n) || n < 0 || n > 10) return '';
      return String(n);
    }

    function isDailyScoreCell_(value) {
      const text = cleanText_(value);
      if (!text) return false;
      if (isDateLike_(text)) return false;

      const n = normalizeVi_(text);
      const rejected = {
        'ma hv': true,
        'ma hoc vien': true,
        'student id': true,
        'id': true,
        'ho ten': true,
        'ten hoc vien': true,
        'student name': true,
        'name': true,
        'daily data': true,
        'diem': true,
        'score': true
      };

      if (rejected[n]) return false;

      return true;
    }

    function parseDailyScoreText_(value) {
      const text = cleanText_(value);

      if (!text) {
        return { score: '--', scoreNumber: '', scoreValue: '', rawScore: '', comment: '' };
      }

      const candidates = findDailyScoreCandidates_(text);

      if (!candidates.length) {
        return {
          score: '--',
          scoreNumber: '',
          scoreValue: '',
          rawScore: '',
          comment: text
        };
      }

      // Ưu tiên điểm phân số như 9/11, 11/11 nếu có trong nhận xét.
      const selected = candidates.find(x => x.type === 'fraction') || candidates[0];
      const comment = buildDailyScoreComment_(text, selected);
      const scoreValue = selected.value;
      const scoreDisplay = selected.type === 'fraction'
        ? formatScoreForDisplay_(roundScoreByCustomRule_(scoreValue))
        : formatScoreForDisplay_(scoreValue);

      return {
        score: scoreDisplay,
        scoreNumber: String(scoreValue),
        scoreValue: scoreValue,
        rawScore: selected.raw,
        comment: comment
      };
    }

    function findDailyScoreCandidates_(text) {
      const candidates = [];
      const occupied = [];
      let m;

      // Dạng phân số: 9/11, 11/11, 10/10...
      const fractionRe = /(\d{1,2}(?:[,.]\d+)?)\s*\/\s*(\d{1,2}(?:[,.]\d+)?)/g;
      while ((m = fractionRe.exec(text)) !== null) {
        const numerator = Number(String(m[1]).replace(',', '.'));
        const denominator = Number(String(m[2]).replace(',', '.'));
        if (isNaN(numerator) || isNaN(denominator) || denominator <= 0) continue;

        const converted = numerator / denominator * 10;
        if (converted < 0 || converted > 10.000001) continue;

        const item = {
          type: 'fraction',
          raw: m[0],
          start: m.index,
          end: m.index + m[0].length,
          value: Math.round(converted * 10000) / 10000
        };
        candidates.push(item);
        occupied.push([item.start, item.end]);
      }

      // Dạng số thường: 10, 9.5, 8,5...
      const decimalRe = /\d{1,2}(?:[,.]\d+)?/g;
      while ((m = decimalRe.exec(text)) !== null) {
        const start = m.index;
        const end = start + m[0].length;
        if (occupied.some(range => start >= range[0] && end <= range[1])) continue;

        const value = Number(String(m[0]).replace(',', '.'));
        if (isNaN(value) || value < 0 || value > 10) continue;

        candidates.push({
          type: 'number',
          raw: m[0],
          start: start,
          end: end,
          value: value
        });
      }

      return candidates.sort((a, b) => a.start - b.start);
    }

    function buildDailyScoreComment_(text, scoreItem) {
      if (!scoreItem) return text;

      const before = text.slice(0, scoreItem.start).trim();
      const after = text.slice(scoreItem.end).trim();

      // Nếu dòng bắt đầu bằng điểm, ví dụ: "10 - bé ngoan" hoặc "9/11 bé tốt",
      // thì bỏ phần điểm khỏi nhận xét để bảng gọn hơn.
      if (!before) {
        return cleanText_(after.replace(/^[\s:：\-–—|,.;]+/g, ''));
      }

      // Nếu điểm nằm trong câu nhận xét, giữ nguyên nhận xét gốc.
      // Ví dụ: "Bé làm tốt (Điểm khảo từ vựng: 9/11)".
      return text;
    }

    function roundScoreByCustomRule_(value) {
      const n = Number(value);
      if (isNaN(n)) return '';

      const base = Math.floor(n);
      const decimal = n - base;

      if (decimal < 0.25) return base;
      if (decimal <= 0.75) return base + 0.5;
      return base + 1;
    }

    function formatScoreForDisplay_(value) {
      const n = Number(value);
      if (isNaN(n)) return '--';

      const fixed = Math.round(n * 100) / 100;
      if (Math.abs(fixed - Math.round(fixed)) < 0.000001) return String(Math.round(fixed));
      return String(fixed).replace('.', ',').replace(',', '.');
    }

    function parseAttendance_(rows) {
      const out = [];

      if (!rows || !rows.length) return out;

      const datePositions = [];
      rows.forEach((row, r) => {
        (row || []).forEach((cell, c) => {
          const date = cleanText_(cell);
          if (isDateLike_(date)) datePositions.push({ r, c, date });
        });
      });

      datePositions.forEach(pos => {
        const status = findAttendanceStatusBelowDate_(rows, pos.r, pos.c);
        if (status) out.push({ date: pos.date, status });
      });

      if (out.length) {
        return sortBaoBaiByDateDesc_(dedupeBaoBaiByDate_(out));
      }

      rows.forEach(row => {
        const cells = (row || []).map(cleanText_);
        const date = cells.find(isDateLike_);
        if (!date) return;

        const status = cells.find(isAttendanceStatusCell_);
        if (!status) return;

        out.push({ date, status });
      });

      return sortBaoBaiByDateDesc_(dedupeBaoBaiByDate_(out));
    }

    function findAttendanceStatusBelowDate_(rows, headerRowIndex, colIndex) {
      for (let r = headerRowIndex + 1; r < Math.min(rows.length, headerRowIndex + 20); r++) {
        const value = cleanText_((rows[r] || [])[colIndex]);
        if (isAttendanceStatusCell_(value)) return value;
      }
      return '';
    }

    function isAttendanceStatusCell_(value) {
      const n = normalizeVi_(value);
      return n.includes('di hoc') ||
        n.includes('dung gio') ||
        n.includes('di tre') ||
        n.includes('tre') ||
        n.includes('vang') ||
        n.includes('co phep') ||
        n.includes('nghi') ||
        n.includes('present') ||
        n.includes('absent') ||
        n.includes('late');
    }

    function buildBaoBaiTimeline_(lessonRows, historyRows) {
      const currentItems = parseBaoBai_(lessonRows || []);
      const historyItems = parseBaoBaiHistory_(historyRows || []);

      // Ưu tiên bản đang nằm trong Baobai nếu cùng ngày cũng đã có
      // trong History-TKB.
      return sortBaoBaiByDateDesc_(
        dedupeBaoBaiByDate_(currentItems.concat(historyItems))
      );
    }

    function parseBaoBaiHistory_(rows) {
      if (!rows || !rows.length) return [];

      const blocks = splitBaoBaiHistoryBlocks_(rows);
      let items = [];

      blocks.forEach(block => {
        items = items.concat(parseBaoBai_(block));
      });

      return sortBaoBaiByDateDesc_(dedupeBaoBaiByDate_(items));
    }

    function splitBaoBaiHistoryBlocks_(rows) {
      const blocks = [];

      for (let rowIndex = 0; rowIndex < rows.length; rowIndex++) {
        const row = rows[rowIndex] || [];
        if (!isBaoBaiHistoryDayHeader_(row)) continue;

        const block = [row];

        // Một block Student Profile gồm 1 dòng Day + 6 dòng nội dung.
        // Dừng sớm nếu gặp block kế tiếp hoặc dòng trống phân cách.
        for (
          let nextIndex = rowIndex + 1;
          nextIndex < rows.length && block.length < 7;
          nextIndex++
        ) {
          const nextRow = rows[nextIndex] || [];
          if (isBaoBaiHistoryDayHeader_(nextRow)) break;

          const hasValue = nextRow.some(value => cleanText_(value));
          if (!hasValue) break;

          block.push(nextRow);
        }

        blocks.push(block);
      }

      return blocks;
    }

    function isBaoBaiHistoryDayHeader_(row) {
      const cells = (row || []).map(cleanText_);
      const firstText = cells.find(Boolean);
      const normalizedFirstText = normalizeVi_(firstText);
      const isDayLabel =
        normalizedFirstText === 'day' ||
        normalizedFirstText === 'ngay';

      return isDayLabel && cells.some(isDateLike_);
    }

    function getBaoBaiDateKey_(value) {
      const text = cleanText_(value);
      if (!text) return 0;

      let match = text.match(/^(\d{1,2})[\/\-](\d{1,2})[\/\-](\d{2,4})$/);
      if (match) {
        const day = Number(match[1]);
        const month = Number(match[2]);
        let year = Number(match[3]);
        if (year < 100) year += 2000;
        return year * 10000 + month * 100 + day;
      }

      match = text.match(/^(\d{4})[\/\-](\d{1,2})[\/\-](\d{1,2})$/);
      if (match) {
        return Number(match[1]) * 10000 +
          Number(match[2]) * 100 +
          Number(match[3]);
      }

      return 0;
    }

    function dedupeBaoBaiByDate_(items) {
      const seen = {};
      const out = [];

      (items || []).forEach(item => {
        const parsedDateKey = getBaoBaiDateKey_(item && item.date);
        const key = parsedDateKey
          ? String(parsedDateKey)
          : normalizeVi_(item && item.date);

        if (!key || seen[key]) return;
        seen[key] = true;
        out.push(item);
      });

      return out;
    }

    function sortBaoBaiByDateDesc_(items) {
      return (items || []).slice().sort((a, b) => {
        return getBaoBaiDateKey_(b && b.date) -
          getBaoBaiDateKey_(a && a.date);
      });
    }

    function parseBaoBai_(rows) {
      if (!rows || !rows.length) return [];

      const horizontal = parseBaoBaiHorizontal_(rows);
      if (horizontal.length) return horizontal;

      return parseBaoBaiRowBased_(rows);
    }

    function parseBaoBaiHorizontal_(rows) {
      const datePositions = [];

      rows.forEach((row, r) => {
        (row || []).forEach((cell, c) => {
          const text = cleanText_(cell);
          if (isDateLike_(text)) {
            datePositions.push({ r, c, date: text });
          }
        });
      });

      if (!datePositions.length) return [];

      const out = [];

      datePositions.forEach(pos => {
        const c = pos.c;
        const material = parseBaoBaiMaterial_(
          findValueInDateColumn_(rows, c, ['tài liệu', 'materials'])
        );

        const item = {
          date: pos.date,
          status: findValueInDateColumn_(rows, c, ['điểm danh', 'trạng thái', 'attendance', 'status']) || 'Đã ghi nhận',
          lessonFocus: findValueInDateColumn_(rows, c, ['chủ đề', 'bài học', 'lesson focus', 'lesson', 'nội dung buổi học']),
          vocabularyGrammar: findValueInDateColumn_(rows, c, ['từ vựng', 'ngữ pháp', 'vocabulary', 'grammar']),
          skillsPracticed: findValueInDateColumn_(rows, c, ['kỹ năng', 'skills']),
          homeworkPractice: findValueInDateColumn_(rows, c, ['btvn', 'bài tập', 'homework']),
          materialsText: material.text,
          materialsUrl: material.url,
          teacherNote: findValueInDateColumn_(rows, c, ['ghi chú', 'nhận xét', 'teacher note', 'note'])
        };

        if (
          item.lessonFocus ||
          item.vocabularyGrammar ||
          item.skillsPracticed ||
          item.homeworkPractice ||
          item.materialsText ||
          item.materialsUrl ||
          item.teacherNote
        ) {
          out.push(item);
        }
      });

      return dedupeByDate_(out).reverse();
    }

    function findValueInDateColumn_(rows, dateCol, labels) {
      const wanted = labels.map(normalizeVi_);

      for (let r = 0; r < rows.length; r++) {
        const row = rows[r] || [];
        // Labels are fixed in column A. Reading only this cell prevents earlier
        // lesson content from matching Grammar, Skills, or Homework by mistake.
        const labelText = normalizeVi_(row[0]);

        const matched = wanted.some(label => labelText.includes(label));
        if (!matched) continue;

        // Đúng nhãn nhưng ô ngày trống vẫn là kết quả hợp lệ.
        return cleanText_(row[dateCol]);
      }

      return '';
    }

    function parseBaoBaiMaterial_(value) {
      const source = cleanText_(value);
      const match = source.match(/https?:\/\/[^\s<>"']+/i);

      if (!match) {
        return { text: source, url: '' };
      }

      const url = match[0];
      const text = cleanText_(
        source
          .replace(url, '')
          .replace(/^[\s:|+\-]+|[\s:|+\-]+$/g, '')
      );

      return { text, url };
    }

    function parseBaoBaiRowBased_(rows) {
      const out = [];

      rows.forEach(row => {
        const cells = (row || []).map(cleanText_).filter(Boolean);
        const date = cells.find(isDateLike_);
        if (!date) return;

        const joined = cells.join('\n');

        out.push({
          date,
          status: cells.find(x => normalizeVi_(x).includes('di hoc') || normalizeVi_(x).includes('vang')) || 'Đã ghi nhận',
          lessonFocus: joined,
          vocabularyGrammar: '',
          skillsPracticed: '',
          homeworkPractice: '',
          materialsText: '',
          teacherNote: ''
        });
      });

      return dedupeByDate_(out).reverse();
    }

    function parseMocktests_(rows, studentId) {
      const out = [];

      if (!rows || !rows.length) return out;

      const targetStudentId = normalizeStudentCode_(studentId);
      let studentRowIndex = -1;

      if (targetStudentId) {
        for (let r = 0; r < rows.length; r++) {
          const rowText = (rows[r] || []).map(cleanText_).join(' ');
          if (normalizeStudentCode_(rowText).includes(targetStudentId)) {
            studentRowIndex = r;
            break;
          }
        }
      }

      // Fallback: chấp nhận bài kiểm tra mới chỉ có điểm ở một kỹ năng.
      if (studentRowIndex < 0) {
        studentRowIndex = rows.findIndex(row => {
          const nums = (row || []).map(cleanText_).filter(isMocktestScoreValue_);
          return nums.length >= 1;
        });
      }

      if (studentRowIndex < 0) return out;

      const studentRow = rows[studentRowIndex] || [];
      const headerRows = rows.slice(0, studentRowIndex);

      const mockHeaders = [];

      headerRows.forEach(row => {
        (row || []).forEach((cell, c) => {
          const text = cleanText_(cell);
          const n = normalizeVi_(text);

          if (
            n.includes('mocktest') ||
            n.includes('mock test') ||
            n.includes('mini test')
          ) {
            mockHeaders.push({
              col: c,
              name: text
            });
          }
        });
      });

      // Fallback theo cấu trúc sheet:
      // A: Mã HV, B: Họ tên, C-E: 1st, F-H: 2nd, I-K: 3rd, L-N: 4th, O-Q: 5th.
      if (!mockHeaders.length) {
        for (let i = 0; i < 5; i++) {
          mockHeaders.push({
            col: 2 + i * 3,
            name: `${i + 1}${getOrdinalSuffix_(i + 1)} Mocktest`
          });
        }
      }

      const seenCols = {};
      const uniqueHeaders = [];

      mockHeaders
        .sort((a, b) => a.col - b.col)
        .forEach(h => {
          if (seenCols[h.col]) return;
          seenCols[h.col] = true;
          uniqueHeaders.push(h);
        });

      uniqueHeaders.forEach((header, idx) => {
        const startCol = header.col;
        const skillLabels = [
          'Listening',
          'Reading & Writing',
          'Speaking (E/G/NI)'
        ];
        const rawItems = [startCol, startCol + 1, startCol + 2]
          .map(col => parseMocktestScore_(studentRow[col]));
        const skillScores = rawItems.map((item, skillIndex) => {
          if (!item) return null;
          return {
            label: skillLabels[skillIndex],
            score: item.display
          };
        }).filter(Boolean);

        if (!skillScores.length) return;

        const scores = skillScores.map(item => item.score);

        const nums = rawItems
          .filter(Boolean)
          .map(item => item.value)
          .filter(n => !isNaN(n));

        const avg = nums.length
          ? Math.round((nums.reduce((a, b) => a + b, 0) / nums.length) * 100) / 100
          : null;

        out.push({
          testName: header.name || `${idx + 1}${getOrdinalSuffix_(idx + 1)} Mocktest`,
          average: avg,
          scores: scores,
          skillScores: skillScores
        });
      });

      return out;
    }

    function parseReportText_(rows) {
      rows = Array.isArray(rows) ? rows : [];

      // Layout chuẩn: nhãn "Nhận xét từ GV" ở bên trái, lời nhắn ở ô gộp bên phải.
      const labeledText = findReportTextBesideLabel_(rows);
      if (labeledText) return labeledText;

      // Fallback cho profile cũ: chọn một ô giống nội dung báo cáo thay vì
      // ghép banner, tiêu đề Mini Test và các nhãn khác của sheet.
      const candidates = [];
      const preferred = [];

      rows.forEach(row => {
        (row || []).forEach(cell => {
          const text = cleanReportText_(cell);
          if (!isReportFallbackCandidate_(text)) return;

          candidates.push(text);
          if (hasReportBodyMarker_(text)) preferred.push(text);
        });
      });

      return pickLongestReportText_(preferred.length ? preferred : candidates);
    }

    function findReportTextBesideLabel_(rows) {
      for (let r = 0; r < rows.length; r++) {
        const row = rows[r] || [];

        for (let c = 0; c < row.length; c++) {
          if (!isReportTextLabel_(row[c])) continue;

          const sameRow = [];
          for (let c2 = c + 1; c2 < row.length; c2++) {
            const text = cleanReportText_(row[c2]);
            if (text.length >= 20 && !isReportTextLabel_(text)) sameRow.push(text);
          }

          if (sameRow.length) return pickLongestReportText_(sameRow);

          const below = [];
          for (let r2 = r + 1; r2 < Math.min(rows.length, r + 3); r2++) {
            const nextRow = rows[r2] || [];
            for (let c3 = Math.max(0, c - 1); c3 < nextRow.length; c3++) {
              const nextText = cleanReportText_(nextRow[c3]);
              if (isReportFallbackCandidate_(nextText)) below.push(nextText);
            }
          }

          if (below.length) return pickLongestReportText_(below);
        }
      }

      return '';
    }

    function isReportTextLabel_(value) {
      const n = normalizeVi_(value);
      return n === 'nhan xet tu gv' ||
        n === 'nhan xet gv' ||
        n === 'nhan xet tu giao vien' ||
        n === 'nhan xet giao vien' ||
        n === 'loi nhan tu giao vien';
    }

    function isReportFallbackCandidate_(text) {
      text = cleanReportText_(text);
      if (text.length < 20) return false;

      const n = normalizeVi_(text);
      if (isReportTextLabel_(text)) return false;
      if (isDateLike_(text) || isScoreValue_(text)) return false;

      if (n.includes('mini test') ||
          n.includes('mock test') ||
          n.includes('mocktest') ||
          n.includes('but pha tieng anh') ||
          n.includes('vuon tam the gioi')) {
        return false;
      }

      return true;
    }

    function hasReportBodyMarker_(text) {
      const n = normalizeVi_(text);
      return n.includes('em lien lac de gui bao cao') ||
        n.includes('diem thi cuoi ky') ||
        n.includes('bao cao tinh hinh hoc tap') ||
        n.includes('ky nang nghe') ||
        n.includes('wewin xin chan thanh cam on');
    }

    function pickLongestReportText_(items) {
      return (items || []).reduce((best, item) => {
        const text = cleanReportText_(item);
        return text.length > best.length ? text : best;
      }, '');
    }

    function cleanReportText_(value) {
      if (value === null || value === undefined) return '';

      return String(value)
        .replace(/\r\n?/g, '\n')
        .replace(/\u00A0/g, ' ')
        .replace(/[ \t]+/g, ' ')
        .replace(/[ \t]*\n[ \t]*/g, '\n')
        .replace(/\n{3,}/g, '\n\n')
        .trim();
    }

    function buildMetrics_(attendance, dailyScores) {
      const present = attendance.filter(x => {
        const s = normalizeVi_(x.status);
        return s.includes('di hoc') ||
          s.includes('dung gio') ||
          s.includes('di tre') ||
          s.includes('tre') ||
          s.includes('di muon') ||
          s.includes('late');
      }).length;
      const absent = attendance.filter(x => normalizeVi_(x.status).includes('vang') || normalizeVi_(x.status).includes('nghi')).length;
      const excused = attendance.filter(x => normalizeVi_(x.status).includes('co phep')).length;

      const nums = (dailyScores || [])
        .map(x => {
          const raw = cleanText_(x.rawScore || x.displayText || '');
          const parsed = parseDailyScoreText_(raw || x.score || '');

          if (parsed.scoreNumber !== '') return Number(parsed.scoreNumber);
          if (x.scoreValue !== undefined && x.scoreValue !== '') return Number(String(x.scoreValue).replace(',', '.'));
          if (x.scoreNumber !== undefined && x.scoreNumber !== '') return Number(String(x.scoreNumber).replace(',', '.'));

          return NaN;
        })
        .filter(n => !isNaN(n));

      const avgRaw = nums.length
        ? nums.reduce((a, b) => a + b, 0) / nums.length
        : '';
      const avg = avgRaw !== '' ? roundScoreByCustomRule_(avgRaw) : '';

      const latestItem = dailyScores && dailyScores.length ? dailyScores[0] : null;
      const latestParsed = latestItem
        ? parseDailyScoreText_(cleanText_(latestItem.rawScore || latestItem.displayText || latestItem.score || ''))
        : null;
      const latest = latestParsed && latestParsed.scoreNumber !== ''
        ? latestParsed.score
        : (latestItem && latestItem.score && latestItem.score !== '--' ? latestItem.score : '--');

      return {
        attendancePresent: present,
        attendanceAbsent: absent,
        attendanceExcused: excused,
        avgScore: avg !== '' ? formatScoreForDisplay_(avg) : '--',
        latestScore: String(latest || '--')
      };
    }

    function cleanScore_(value) {
      const parsed = parseMocktestScore_(value);
      return parsed ? parsed.display : '';
    }

    function parseMocktestScore_(value) {
      const display = cleanText_(value);
      if (!display) return null;

      const normalized = display.replace(',', '.');
      const fraction = normalized.match(/^(-?\d+(?:\.\d+)?)\s*\/\s*(-?\d+(?:\.\d+)?)$/);

      if (fraction) {
        const numerator = Number(fraction[1]);
        const denominator = Number(fraction[2]);

        if (isNaN(numerator) || isNaN(denominator) || numerator < 0 || denominator < 0) return null;
        if (denominator === 0) {
          return numerator === 0 ? { display, value: NaN } : null;
        }
        if (numerator > denominator) return null;

        return {
          display,
          value: Math.round((numerator / denominator * 10) * 100) / 100
        };
      }

      if (!/^-?\d+(\.\d+)?$/.test(normalized)) return null;

      const n = Number(normalized);
      if (isNaN(n) || n < 0 || n > 100) return null;

      return {
        display,
        value: n
      };
    }

    function isMocktestScoreValue_(value) {
      return Boolean(parseMocktestScore_(value));
    }

    function isScoreValue_(value) {
      const text = cleanText_(value).replace(',', '.');

      if (!/^-?\d+(\.\d+)?$/.test(text)) return false;

      const n = Number(text);
      return n >= 0 && n <= 10;
    }

    function isDateLike_(value) {
      value = cleanText_(value);

      if (!value) return false;

      return /^\d{1,2}[\/\-]\d{1,2}[\/\-]\d{2,4}$/.test(value) ||
        /^\d{4}[\/\-]\d{1,2}[\/\-]\d{1,2}$/.test(value);
    }

    function mergeTimelineLiveFirst_(liveItems, historyItems) {
      return dedupeByDate_((liveItems || []).concat(historyItems || []))
        .sort((a, b) => getBaoBaiDateKey_(b && b.date) - getBaoBaiDateKey_(a && a.date));
    }
    function dedupeByDate_(items) {
      const seen = {};
      const out = [];

      items.forEach(item => {
        const key = cleanText_(item.date);
        if (!key || seen[key]) return;
        seen[key] = true;
        out.push(item);
      });

      return out;
    }

    function extractStudentCodeFromText_(value) {
      const text = cleanText_(value);
      const match = text.match(/[A-Za-z0-9]+(?:-[A-Za-z0-9]+)*-(?:HV|KH|HS)-\d+/i);
      return match && match[0] ? match[0].toUpperCase() : '';
    }

    function normalizeStudentCode_(value) {
      return cleanText_(value)
        .toUpperCase()
        .replace(/\s+/g, '');
    }

    function getOrdinalSuffix_(n) {
      if (n === 1) return 'st';
      if (n === 2) return 'nd';
      if (n === 3) return 'rd';
      return 'th';
    }

    function cleanText_(value) {
      if (value === null || value === undefined) return '';

      return String(value)
        .replace(/\u00A0/g, ' ')
        .replace(/\s+/g, ' ')
        .trim();
    }

    function normalizeVi_(value) {
      return cleanText_(value)
        .toLowerCase()
        .normalize('NFD')
        .replace(/[\u0300-\u036f]/g, '')
        .replace(/đ/g, 'd');
    }

    function attCls_(s) {
      s = normalizeVi_(s || '');
      if (s.includes('di hoc') || s.includes('dung gio')) return 'sp-ok';
      if (s.includes('di tre') || s.includes('tre') || s.includes('di muon') || s.includes('late')) return 'sp-warn';
      if (s.includes('vang')) return 'sp-err';

    }

    function renderBaoBaiSection_(items) {
      const root = document.getElementById('baoBaiList');
      if (!items.length) {
        document.getElementById('baobaiDayTitle').textContent = '--';
        root.innerHTML = `<div class="bb-empty">${uiIcon_('inbox')} Chưa có báo bài nào được ghi nhận</div>`;

      }
      if (window.__baobaiIdx === undefined) window.__baobaiIdx = 0;

      const item = items[window.__baobaiIdx];

    function renderBaoBaiCard_(item) {
      // ── 1. Attendance status class ──
      const sNorm = normalizeVi_(item.status || '');
      const sc = (sNorm.includes('di hoc') || sNorm.includes('dung gio')) ? 'bb-sp' :
                 (sNorm.includes('di tre') || sNorm.includes('tre') || sNorm.includes('di muon') || sNorm.includes('late')) ? 'bb-sw' :
                 sNorm.includes('vang') ? 'bb-sa' : 'bb-sn';
      const statusLabel = escapeHtml_(item.status || 'Đã ghi nhận');

      // ── 2. Header: lesson headline (2-line clamp) ──
      const lessonForHead = item.lessonFocus
        ? `<span class="bb-head-lesson">${escapeHtml_(item.lessonFocus)}</span>`
        : `<span class="bb-head-lesson is-empty">Chưa có tiêu đề buổi học</span>`;

      const header = `
        <div class="bb-head">
          <div class="bb-head-main">
            <div class="bb-head-eyebrow">CHỦ ĐỀ BUỔI HỌC</div>
            ${lessonForHead}
          </div>
          <div class="bb-status ${sc}">${statusLabel}</div>
        </div>`;

      // ── 3. Body content ──
      let bodyContent;

      if (item.locked) {
        // Locked: chỉ hiện lock box, tuyệt đối không render field nào
        bodyContent = `
          <div class="bb-lock-box">
            <div class="bb-lock-icon">${uiIcon_('lock')}</div>
            <div class="bb-lock-title">Nội dung chưa được mở khóa</div>
            <div class="bb-lock-desc">Giáo viên sẽ cập nhật nội dung buổi học này sau khi buổi học diễn ra.</div>
          </div>`;
      } else {
        // We removed Tier 1 (Lesson Focus) since it's already in the header to avoid repetition
        
        // Tier 2: Vocab + Skills (2-col grid, only if at least one exists)
        const hasVocab = Boolean(item.vocabularyGrammar);
        const hasSkills = Boolean(item.skillsPracticed);
        let t2 = '';
        if (hasVocab || hasSkills) {
          const vocabBlock = hasVocab ? `
            <div class="bb-sec-item">
              <div class="bb-item-lbl">${uiIcon_('book-open')} Từ vựng &amp; Ngữ pháp</div>
              <div class="bb-value">${nl2br_(escapeHtml_(item.vocabularyGrammar))}</div>
            </div>` : '';
          const skillsBlock = hasSkills ? `
            <div class="bb-sec-item">
              <div class="bb-item-lbl">${uiIcon_('activity')} Kỹ năng luyện tập</div>
              <div class="bb-value">${nl2br_(escapeHtml_(item.skillsPracticed))}</div>
            </div>` : '';
          // If only one exists, use single-col naturally; grid handles it
          t2 = `<div class="bb-secondary-grid">${vocabBlock}${skillsBlock}</div>`;
        }

        // Tier 3: Homework (highlighted)
        const t3 = item.homeworkPractice ? `
          <div class="bb-homework">
            <div class="bb-item-lbl">✏️ Bài tập về nhà</div>
            <div class="bb-value">${nl2br_(escapeHtml_(item.homeworkPractice))}</div>
          </div>` : '';

        // Tier 4: Materials
        const t4 = bbMaterials_(item);

        // Tier 5: Teacher note
        const t5 = item.teacherNote ? `
          <div class="bb-teacher-note">
            <div class="bb-tn-label">${uiIcon_('message')} Ghi chú từ giáo viên</div>
            <div class="bb-tn-text">${nl2br_(escapeHtml_(item.teacherNote))}</div>
          </div>` : '';

        const hasContent = t2 || t3 || t4 || t5;
        bodyContent = hasContent
          ? [t2, t3, t4, t5].filter(Boolean).join('\n')
          : `<div class="bb-empty" style="padding:24px 0;">${uiIcon_('clipboard')} Chưa có nội dung chi tiết cho buổi học này</div>`;
      }

      return `
        <article class="bb-card">
          ${header}
          <div class="bb-body">${bodyContent}</div>
        </article>`;
    }

    // Helper: materials block (kept for compatibility alias below)
    function bbMaterials_(item) {
      let text = (item && item.materialsText ? String(item.materialsText) : '').trim();
      let url = (item && item.materialsUrl ? String(item.materialsUrl) : '').trim();

      // Tương thích dữ liệu cũ: URL từng được trả về trong materialsText.
      if (!url && text) {
        const material = parseBaoBaiMaterial_(text);
        text = material.text;
        url = material.url;
      }

      if (!text && !url) return '';
      const label = text || 'Mở tài liệu';

        ? `<a href="${escapeAttr_(url)}" target="_blank" rel="noopener noreferrer" class="bb-link">${uiIcon_('link')} ${escapeHtml_(label)}</a>`
        : `<span style="font-size:13.5px;font-weight:500;color:var(--text);">${nl2br_(escapeHtml_(label))}</span>`;
      return `<div class="bb-materials">${inner}</div>`;
    }

    function getVisibleMocktestSkills_(item) {
      const fallbackLabels = ['Listening', 'Reading & Writing', 'Speaking (E/G/NI)'];
      const provided = Array.isArray(item && item.skillScores) ? item.skillScores : [];
      const source = provided.length
        ? provided.map((entry, idx) => {
            const score = entry && typeof entry === 'object'
              ? (entry.score !== undefined ? entry.score : entry.value)
              : entry;
            return {
              label: cleanText_(entry && entry.label) || fallbackLabels[idx] || '',
              score: cleanText_(score)
            };
          })
        : ((item && item.scores) || []).map((score, idx) => ({
            label: fallbackLabels[idx] || '',
            score: cleanText_(score)
          }));

      return source.filter(entry => !isMissingMocktestScore_(entry.score));
    }

    /** Dòng nhãn trống sau dấu ":" (vd. "• Kỹ năng nghe:") — ẩn trên UI. */
    function isEmptyColonLabelLine_(line) {
      const t = String(line || '').trim();
      if (!t) return false;

      // Bullet / list label với không có gì sau dấu :
      if (/^[•\-]\s*.+:\s*$/.test(t)) return true;

      // Nhãn kỹ năng không có bullet
      if (/^kỹ năng[^:\n]*:\s*$/iu.test(t)) return true;

      return false;
    }

    function isMissingReportScoreLine_(line) {
      const withoutBullet = String(line || '').replace(/^[\u2022\-]\s*/, '').trim();
      const normalized = normalizeVi_(withoutBullet);
      const match = normalized.match(/^(reading\s*(?:&|and)\s*writing|reading|writing|listening|speaking(?:\s*\([^)]*\))?|nghe|doc|viet|noi)\s*:\s*(.*)$/);

      if (!match) return false;

      const scoreText = String(match[2] || '').replace(/\s+/g, ' ').trim();
      return !scoreText ||
        scoreText === 'chua co diem' ||
        scoreText === 'chua co' ||
        scoreText === 'chua cap nhat' ||
        scoreText === 'chua thi' ||
        scoreText === '--' ||
        scoreText === 'n/a' ||
        scoreText === 'na';
    }
    function formatReportText_(text) {
      let s = String(text || '')
        .replace(/\r\n?/g, '\n')
        .replace(/\u00A0/g, ' ')
        .trim();

      if (!s) return '';

      // Chỉ đổi dấu gạch đầu dòng khi có khoảng trắng quanh dấu "-",
      // để không làm hỏng các từ như "nói-phát âm", "đọc-viết".
      s = s.replace(/(^|\n)\s*[-•]\s+/g, '$1• ');
      s = s.replace(/([^\n])\s+-\s+/g, '$1\n• ');

      // Ép các tiêu đề/đoạn kết quan trọng xuống dòng riêng.
      // Đây là phần đang làm chữ bị dính như: "... ạ 📝 Điểm thi...".
      const sectionRules = [
        [/\s*(📝\s*Điểm thi cuối kỳ\s*:)/gu, '\n\n$1'],
        [/\s*(✍️?\s*Báo cáo tình hình học tập\s*:)/gu, '\n\n$1'],
        [/\s*(📣\s*Trong thời gian tới)/gu, '\n\n$1'],
        [/\s*(💟\s*WEWIN xin chân thành cảm ơn)/gu, '\n\n$1'],
        [/\s*(💟\s*WEWIN Education)/gu, '\n$1']
      ];

      sectionRules.forEach(([pattern, replacement]) => {
        s = s.replace(pattern, replacement);
      });

      // Ẩn dòng kỹ năng/nhãn bullet không có nội dung sau dấu :
      s = s
        .split('\n')
        .filter(line => !isEmptyColonLabelLine_(line))
        .filter(line => !isMissingReportScoreLine_(line))
        .join('\n');

      return s
        .replace(/[ \t]+\n/g, '\n')
        .replace(/\n[ \t]+/g, '\n')
        .replace(/\n{3,}/g, '\n\n')
        .trim();
    }

    function renderReportText_(text, hasPerTestFeedback) {

    }

    function renderLearningPath_(className) {
      const wrap = document.getElementById('learningPathWrap');
      if (!wrap) return;

      const levels = [
        { id: 'starters', name: 'Starters', icon: 'star' },
        { id: 'movers', name: 'Movers', icon: 'book-open' },
        { id: 'flyers', name: 'Flyers', icon: 'send' },
        { id: 'ket', name: 'KET', icon: 'clipboard' },
        { id: 'explorer', name: 'IELTS Explorer', icon: 'globe' },
        { id: 'pre', name: 'Pre-IELTS', icon: 'star' },
        { id: 'foundation', name: 'IELTS Foundation', icon: 'chart' },
        { id: 'intensive', name: 'IELTS Intensive', icon: 'target' }
      ];

      let currentLevelIndex = -1;
      const classNorm = normalizeVi_(className || '').toUpperCase();
      
      if (classNorm.includes('INTENSIVE')) currentLevelIndex = 7;
      else if (classNorm.includes('FOUNDATION')) currentLevelIndex = 6;
      else if (classNorm.includes('PRE-IELTS') || classNorm.includes('PRE IELTS') || classNorm.includes('PREIELTS')) currentLevelIndex = 5;
      else if (classNorm.includes('EXPLORER')) currentLevelIndex = 4;
      else if (classNorm.includes('KET')) currentLevelIndex = 3;
      else if (classNorm.includes('FLYER')) currentLevelIndex = 2;
      else if (classNorm.includes('MOVER')) currentLevelIndex = 1;
      else if (classNorm.includes('STARTER')) currentLevelIndex = 0;

      let html = '<div class="lp-container"><div class="lp-wrapper"><div class="lp-global-line"></div>';
      
      levels.forEach((lvl, idx) => {
        let statusClass = '';
        let isPassed = false;
        let isActive = false;
        if (currentLevelIndex !== -1) {
          if (idx < currentLevelIndex) { statusClass = 'passed'; isPassed = true; }
          else if (idx === currentLevelIndex) { statusClass = 'active'; isActive = true; }
        }

        const height = 90 + (idx * 20); // Height increments to create the staircase feel
        
        html += `
          <div class="lp-step-col">
            <div class="lp-card ${statusClass}" style="height: ${height}px;">
              <div class="lp-number">${idx + 1}</div>
              <div class="lp-icon">${uiIcon_(lvl.icon)}</div>
              <div class="lp-title">${escapeHtml_(lvl.name)}</div>
              ${isActive ? '<div class="lp-current-badge"><svg class="ui-icon ui-icon-star" aria-hidden="true" focusable="false"><use href="#i-star"></use></svg> Bạn đang ở đây</div>' : ''}
            </div>
            <div class="lp-dot-area">
              <div class="lp-dot ${statusClass}"></div>
            </div>
          </div>
        `;
        if (idx < levels.length - 1) {
          html += `<div class="lp-arrow"><svg class="ui-icon ui-icon-arrow-right" aria-hidden="true" focusable="false"><use href="#i-arrow-right"></use></svg></div>`;
        }
      });
      html += '</div></div>';
      
      wrap.innerHTML = html;
    }
